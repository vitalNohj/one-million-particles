18 |       if (interval) await new Promise((resolve) => setTimeout(resolve, interval))

/packages/eslint-plugin/docs/rules/no-new-in-loop.md:
--------------------------------------------------------------------------------
 1 | Instantiating new objects in the frame loop can waste large amounts of memory,
 2 | which is especially bad for large CPU containers such as Three.js classes and any GPU resource as there is no reliable garbage collection.
 3 | Instead create once in a `useMemo` or a single shared reference outside of the component.
 4 | 
 5 | #### ❌ Incorrect
 6 | 
 7 | This creates a new vector 60+ times a second allocating large amounts of memory.
 8 | 
 9 | ```js
10 | function MoveTowards({ x, y, z }) {
11 |   const ref = useRef()
12 | 
13 |   useFrame(() => {
14 |     ref.current.position.lerp(new THREE.Vector3(x, y, z), 0.1)
15 |   })
16 | 
17 |   return <mesh ref={ref} />
18 | }
19 | ```
20 | 
21 | #### ✅ Correct
22 | 
23 | This creates a vector outside of the frame loop to be reused each frame.
24 | 
25 | ```js
26 | const tempVec = new THREE.Vector3()
27 | 
28 | function MoveTowards({ x, y, z }) {
29 |   const ref = useRef()
30 | 
31 |   useFrame(() => {
32 |     ref.current.position.lerp(tempVec.set(x, y, z), 0.1)
33 |   })
34 | 
35 |   return <mesh ref={ref} />
36 | }
37 | ```
38 | 
39 | This creates a vector once outside of the frame loop inside a `useMemo` to be reused each frame.
40 | 
41 | ```js
42 | function MoveTowards({ x, y, z }) {
43 |   const ref = useRef()
44 |   const tempVec = useMemo(() => new THREE.Vector3())
45 | 
46 |   useFrame(() => {
47 |     ref.current.position.lerp(tempVec.set(x, y, z), 0.1)
48 |   })
49 | 
50 |   return <mesh ref={ref} />
51 | }
52 | ```

15 | } as const

17 | /** Default R3F event manager for web */
18 | export function createPointerEvents(store: RootStore): EventManager<HTMLElement> {
19 |   const { handlePointer } = createEvents(store)
20 | 
21 |   return {
22 |     priority: 1,
23 |     enabled: true,
24 |     compute(event: DomEvent, state: RootState, previous?: RootState) {
25 |       // https://github.com/pmndrs/react-three-fiber/pull/782
26 |       // Events trigger outside of canvas when moved, use offsetX/Y by default and allow overrides
27 |       state.pointer.set((event.offsetX / state.size.width) * 2 - 1, -(event.offsetY / state.size.height) * 2 + 1)
28 |       state.raycaster.setFromCamera(state.pointer, state.camera)
29 |     },
30 | 
31 |     connected: undefined,
32 |     handlers: Object.keys(DOM_EVENTS).reduce(
33 |       (acc, key) => ({ ...acc, [key]: handlePointer(key) }),
34 |       {},
35 |     ) as unknown as Events,
36 |     update: () => {
37 |       const { events, internal } = store.getState()
38 |       if (internal.lastEvent?.current && events.handlers) events.handlers.onPointerMove(internal.lastEvent.current)
39 |     },
40 |     connect: (target: HTMLElement) => {
41 |       const { set, events } = store.getState()
42 |       events.disconnect?.()
43 |       set((state) => ({ events: { ...state.events, connected: target } }))
44 |       if (events.handlers) {
45 |         for (const name in events.handlers) {
46 |           const event = events.handlers[name as keyof typeof events.handlers]
47 |           const [eventName, passive] = DOM_EVENTS[name as keyof typeof DOM_EVENTS]
48 |           target.addEventListener(eventName, event, { passive })
49 |         }
50 |       }
51 |     },
52 |     disconnect: () => {
53 |       const { set, events } = store.getState()
54 |       if (events.connected) {
55 |         if (events.handlers) {
56 |           for (const name in events.handlers) {
57 |             const event = events.handlers[name as keyof typeof events.handlers]
58 |             const [eventName] = DOM_EVENTS[name as keyof typeof DOM_EVENTS]
59 |             events.connected.removeEventListener(eventName, event)
60 |           }
61 |         }
62 |         set((state) => ({ events: { ...state.events, connected: undefined } }))
63 |       }

 2 | title: Basic Animations
 3 | description: This guide will help you understand refs, useFrame and how to make basic animations with Fiber
 4 | nav: 17
 5 | ---
 6 | 
 7 | This tutorial will assume some React knowledge, and will be based on [this starter codesandbox](https://codesandbox.io/s/getting-started-01-12q81?from-embed), so just fork it and follow along!
 8 | 
 9 | We will build a really small, continuous animation loop, that will be the basic building block of more advanced animations later on.
10 | 
11 | ## `useFrame`
12 | 
13 | `useFrame` is a Fiber hook that lets you execute code on every frame of Fiber's render loop. This can have a lot of uses, but we will focus on building an animation with it.
14 | 
15 | It's important to remember that **Fiber hooks can only be called inside a `<Canvas />` parent**!
16 | 
17 | ```jsx
18 | import { useFrame } from '@react-three/fiber'
19 | 
20 | function MyAnimatedBox() {
21 |   useFrame(() => {
22 |     console.log("Hey, I'm executing every frame!")
23 |   })
24 |   return (
25 |     <mesh>
26 |       <boxGeometry />
27 |       <meshBasicMaterial color="royalblue" />
28 |     </mesh>
29 |   )
30 | }
31 | ```
32 | 
33 | This loop is the basic building block of our animation, the callback we pass to `useFrame` will be executed every frame and it will be passed an object containing the state of our Fiber scene:
34 | 
35 | For example, we can extract time information from the `clock` parameter, to know how much time has elapsed in our application, and use that time to animate a value:
36 | 
37 | ```jsx
38 | useFrame(({ clock }) => {
39 |   const a = clock.elapsedTime
40 |   console.log(a) // the value will be 0 at scene initialization and grow each frame
41 | })
42 | ```
43 | 
44 | `clock` is a [three.js Clock](https://threejs.org/docs/#api/en/core/Clock) object, from which we are getting the total elapsed time, which will be key for our animations.
45 | 
46 | ## Animating with Refs
47 | 
48 | It would be tempting to just update the state of our component via `setState` and let it change the `mesh` via props, but going through state isn't ideal, when dealing with continuous updates, commonly know as [transient updates]().
49 | Instead, we want to **directly mutate our mesh each frame**. First, we'll have to get a `reference` to it, via the `useRef` React hook:
50 | 
51 | ```jsx
52 | import React from 'react'
53 | 
54 | function MyAnimatedBox() {
55 |   const myMesh = React.useRef()
56 |   return (
57 |     <mesh ref={myMesh}>
58 |       <boxGeometry />
59 |       <meshBasicMaterial color="royalblue" />
60 |     </mesh>
61 |   )
62 | }
63 | ```
64 | 
65 | `myMesh` will now hold a reference to the actual three.js object, which we can now freely mutate in `useFrame`, without having to worry about React:
66 | 
67 | ```jsx
68 | useFrame(({ clock }) => {
69 |   myMesh.current.rotation.x = clock.elapsedTime
70 | })
71 | ```
72 | 
73 | Let's have a closer look:
74 | 
75 | - We are destructuring `clock` from the argument passed to `useFrame`, which we know is the state of our Fiber scene.
76 | - We are accessing the `rotation.x` property of `myMesh.current` object, which is a reference to our mesh object
77 | - We are assigning our time-dependent value `a` to the `rotation` on the `x` axis, meaning our object will now infinitely rotate between -1 and 1 radians around the x axis!
78 | 
79 | <Codesandbox id="29gxw" />
80 | 
81 | **Exercises**
82 | 
83 | - Try `Math.sin(clock.elapsedTime)` and see how your animation changes
84 |

88 | 
89 | If you want to go deeper into animations, check these out:
90 |

 23 | ## `ReactThreeTestInstance` ⚛️
 24 | 
 25 | This is an internal class that wraps the elements returned from [`ReactThreeTestRenderer.create`](/packages/test-renderer/markdown/rttr.md#create). It has several properties & methods to enhance the testing experience. Similar to the core API, it closely mirrors the API of [`react-test-renderer`](https://reactjs.org/docs/test-renderer.html).
 26 | 
 27 | ### `instance` <a id="instance-prop-instance"></a>
 28 | 
 29 | ```ts
 30 | testInstance.instance
 31 | ```
 32 | 
 33 | Returns the instance object of the specific testInstance. This will be the `THREE` initialized class.
 34 | 
 35 | ### `type` <a id="instance-prop-type"></a>
 36 | 
 37 | ```ts
 38 | testInstance.type
 39 | ```
 40 | 
 41 | Returns the `THREE` type of the test instance, e.g `Scene` or `Mesh`.
 42 | 
 43 | ### `props` <a id="instance-prop-props"></a>
 44 | 
 45 | ```ts
 46 | testInstance.props
 47 | ```
 48 | 
 49 | Returns an object of the props that are currently being passed to the element. This will include hidden ones such as `attach="geometry"` which are automatically applied in the reconciler.
 50 | 
 51 | ### `parent` <a id="instance-prop-parent"></a>
 52 | 
 53 | ```ts
 54 | testInstance.parent
 55 | ```
 56 | 
 57 | Returns the parent testInstance of this testInstance. If no parent is available, it will return `null`.
 58 | 
 59 | ### `children` <a id="instance-prop-children"></a>
 60 | 
 61 | ```ts
 62 | testInstance.children
 63 | ```
 64 | 
 65 | Returns the children test instances of this test instance according to the property `children`, this will not include Geometries, Materials etc.
 66 | 
 67 | ### `allChildren` <a id="instance-prop-allChildren"></a>
 68 | 
 69 | ```ts
 70 | testInstance.allChildren
 71 | ```
 72 | 
 73 | Returns all the children testInstances of this test instance, this will be as thorough as [`testRenderer.toTree()`](/packages/test-renderer/markdown/rttr.md#create-totree) capturing all react components in the tree.
 74 | 
 75 | ### `find()` <a id="instance-meth-find"></a>
 76 | 
 77 | ```ts
 78 | testInstance.find(test)
 79 | ```
 80 | 
 81 | Find a single test instance for which `test(testInstance)` returns `true`. If `test(testInstance)` does not return `true` for exactly one test instance it will throw an error.

 84 | 
 85 | ```ts
 86 | testInstance.findAll(test)
 87 | ```
 88 |

  6 | 
  7 | Like with every other application testing is an important factor when it comes to releasing an application into the wild and when it comes to React Three Fiber we can use React Three Test Renderer to achieve this.
  8 | 
  9 | We will be testing the [sandbox](https://codesandbox.io/s/98ppy) we created in [events and interactions](events-and-interaction).
 10 | 
 11 | ## How to test React Three Fiber
 12 | 
 13 | Let's start by installing the React Three Test Renderer:
 14 | 
 15 | ```bash
 16 | npm install @react-three/test-renderer --save-dev
 17 | ```
 18 | 
 19 | Afterwards, if you are using Create React App you can just add a file that ends in `.test.js` and start writing your code, because React Three Test Renderer is testing library agnostic, so it works with libraries such as `jest`, `jasmine` etc.
 20 | 
 21 | Let's create an `App.test.js` and set up all our test cases:
 22 | 
 23 | ```jsx
 24 | import ReactThreeTestRenderer from '@react-three/test-renderer'
 25 | import { MyRotatingBox } from './App'
 26 | 
 27 | test('mesh to have two children', async () => {
 28 |   const renderer = await ReactThreeTestRenderer.create(<MyRotatingBox />)
 29 | })
 30 | 
 31 | test('click event makes box bigger', async () => {
 32 |   const renderer = await ReactThreeTestRenderer.create(<MyRotatingBox />)
 33 | })
 34 | ```
 35 | 
 36 | In here we created three tests and in each we made sure we created the renderer by using the `create` function.
 37 | 
 38 | Let's start with the first test and make sure our mesh has two children, the material and cube.
 39 | 
 40 | We can start by getting the scene and it's children from the test instance we just created like so:
 41 | 
 42 | ```js
 43 | const meshChildren = renderer.scene.children
 44 | ```
 45 | 
 46 | If you log this mesh out you can see that it returns an array of one element since that's all we have in the scene.
 47 | 
 48 | Using this we can make sure to get that first child and use the `allChildren` property on it like so:
 49 | 
 50 | ```js
 51 | const meshChildren = renderer.scene.children[0].allChildren
 52 | ```
 53 | 
 54 | There is also one property called `children` but this one is meant to be used for things like groups as this one does not return the geometry and the materials, for that we need `allChildren`.
 55 | 
 56 | Now to create our assertion:
 57 | 
 58 | ```js
 59 | expect(meshChildren.length).toBe(2)
 60 | ```
 61 | 
 62 | Our first test case looks like this:
 63 | 
 64 | ```js
 65 | test('mesh to have two children', async () => {
 66 |   const renderer = await ReactThreeTestRenderer.create(<MyRotatingBox />)
 67 |   const mesh = renderer.scene.children[0].allChildren
 68 |   expect(mesh.length).toBe(2)
 69 | })
 70 | ```
 71 | 
 72 | ## Testing interactions
 73 | 
 74 | Now that we have gotten the first test out of the way we can test our interaction and make sure that when we click on the mesh it does indeed update the scale.
 75 | 
 76 | We can do that by utilizing the `fireEvent` method existing in a test instance.
 77 | 
 78 | We know we can get the mesh with:
 79 | 
 80 | ```js
 81 | const mesh = renderer.scene.children[0]
 82 | ```
 83 | 
 84 | Since we already have that we can fire an event in it like so:
 85 | 
 86 | ```js
 87 | await renderer.fireEvent(mesh, 'click')
 88 | ```
 89 | 
 90 | With that done, all that's left to do is the tree demonstration of our scene and make sure the scale prop on our mesh has updated:
 91 | 
 92 | ```js
 93 | expect(mesh.props.scale).toBe(1.5)
 94 | ```
 95 | 
 96 | In the end our test looks something like this:
 97 | 
 98 | ```js
 99 | test('click event makes box bigger', async () => {
100 |   const renderer = await ReactThreeTestRenderer.create(<MyRotatingBox />)
101 |   const mesh = renderer.scene.children[0]
102 |   expect(mesh.props.scale).toBe(1)
103 |   await renderer.fireEvent(mesh, 'click')
104 |   expect(mesh.props.scale).toBe(1.5)
105 | })
106 | ```

  9 | ## Using `TextureLoader` and `useLoader`
 10 | 
 11 | To load the textures we will use the `TextureLoader` from three.js in combination with `useLoader` that will allow us to pass the location of the texture and get the map back.
 12 | 
 13 | It's better to explain with code, let's say you downloaded [this texture](https://cc0textures.com/view?id=PavingStones092) and placed it in the public folder of your site, to get the color map from it you could do:
 14 | 
 15 | ```js
 16 | const colorMap = useLoader(TextureLoader, 'PavingStones092_1K_Color.jpg')
 17 | ```
 18 | 
 19 | Let's then with this information create a small scene where we can use this texture:
 20 | 
 21 | ```jsx
 22 | import { Suspense } from 'react'
 23 | import { Canvas, useLoader } from '@react-three/fiber'
 24 | import { TextureLoader } from 'three'
 25 | 
 26 | function Scene() {
 27 |   const colorMap = useLoader(TextureLoader, 'PavingStones092_1K_Color.jpg')
 28 |   return (
 29 |     <>
 30 |       <ambientLight intensity={0.2} />
 31 |       <directionalLight />
 32 |       <mesh>
 33 |         <sphereGeometry args={[1, 32, 32]} />
 34 |         <meshStandardMaterial />
 35 |       </mesh>
 36 |     </>
 37 |   )
 38 | }
 39 | 
 40 | export default function App() {
 41 |   return (
 42 |     <Canvas>
 43 |       <Suspense fallback={null}>
 44 |         <Scene />
 45 |       </Suspense>
 46 |     </Canvas>
 47 |   )
 48 | }
 49 | ```
 50 | 
 51 | If everything went according to plan, you should now be able to apply this texture to the sphere like so:
 52 | 
 53 | ```jsx
 54 | <meshStandardMaterial map={colorMap} />
 55 | ```

 58 |

 60 | 
 61 | ```js
 62 | const [colorMap, displacementMap, normalMap, roughnessMap, aoMap] = useLoader(TextureLoader, [
 63 |   'PavingStones092_1K_Color.jpg',
 64 |   'PavingStones092_1K_Displacement.jpg',
 65 |   'PavingStones092_1K_Normal.jpg',
 66 |   'PavingStones092_1K_Roughness.jpg',
 67 |   'PavingStones092_1K_AmbientOcclusion.jpg',
 68 | ])
 69 | ```

 74 | <meshStandardMaterial

 18 | ## `create()` 🧪 <a id="create"></a>
 19 | 
 20 | ```tsx
 21 | const renderer = ReactThreeTestRenderer.create(element, options)
 22 | ```
 23 | 
 24 | Create a ReactThreeTestRenderer instance with a passed `three` element e.g. `<mesh />`. By default, it won't create an actual `THREE.WebGLRenderer` and there will be no loop. But it will still render the complete scene graph. Returns the properties below.
 25 | 
 26 | #### CreateOptions
 27 | 
 28 | ```ts
 29 | // RenderProps is from react-three-fiber
 30 | interface CreateOptions extends RenderProps<HTMLCanvasElement> {

 32 |   height?: number // height of canvas
 33 | }
 34 | ```

 36 | ### `scene` <a id="create-scene"></a>
 37 | 
 38 | ```tsx
 39 | renderer.scene
 40 | ```
 41 | 
 42 | Returns the root “react three test instance” object that is useful for making assertions. You can use it to find other “test instances” deeper below.
 43 | 
 44 | ### `getInstance()` <a id="create-getinstance"></a>
 45 | 
 46 | ```tsx
 47 | renderer.getInstance()
 48 | ```
 49 | 
 50 | Return the instance corresponding to the root three element, if available. This will not work if the root element is a function component because they don’t have instances.
 51 | 
 52 | ### `toTree()` <a id="create-totree"></a>
 53 | 
 54 | ```tsx
 55 | renderer.toTree()
 56 | ```
 57 | 
 58 | Returns an object representing the rendered tree similar to [`react-test-renderer`](https://reactjs.org/docs/test-renderer.html#overview). This will include all elements written as react components.
 59 | 
 60 | ### `toGraph()` <a id="create-tograph"></a>
 61 | 
 62 | ```tsx
 63 | renderer.toGraph()
 64 | ```
 65 | 
 66 | Returns an object representing the [`scene graph`](https://threejs.org/manual/#en/scenegraph). This will not include all elements such as ones that use `attach`.

 68 | ### `fireEvent()` <a id="create-fireevent"></a>
 69 | 
 70 | ```tsx
 71 | renderer.fireEvent(testInstance, eventName, mockEventData)
 72 | ```
 73 | 
 74 | Native method to fire events on the specific part of the rendered tree through passing an element within the tree and an event name. The third argument is appended to the [`MockSyntheticEvent`](#create-fireevent-mocksyntheticevent) passed to the event handler.
 75 | 
 76 | Event names follow camelCase convention (e.g. `pointerUp`), or you can pass event handler name instead (e.g. `onPointerUp`).

 92 | 
 93 | ```tsx
 94 | renderer.advanceFrames(frames, delta)
 95 | ```
 96 | 
 97 | Native method to advance the frames (therefore running subscribers to the GL Render loop such as `useFrame`). Requires an amount of frames to advance by & a parameter of delta to pass to the subscribers creating a more controlled testing environment.

100 | 
101 | ```tsx
102 | renderer.update(element)
103 | ```
104 | 
105 | Rerender the tree with the new root element. This simulates a react update at the update, thus updating the children below. If the new element has the same type and key as the previous element, the tree will be updated.

  7 | ## Typing with `useRef`
  8 | 
  9 | React's `useRef` won't automatically infer types despite pointing it to a typed ref.
 10 | 
 11 | You can type the ref yourself by passing a type through `useRef`'s generics:
 12 | 
 13 | ```tsx
 14 | import { useRef, useEffect } from 'react'
 15 | import { Mesh } from 'three'
 16 | 
 17 | function Box(props) {
 18 |   const meshRef = useRef<Mesh>(null!)
 19 | 
 20 |   useEffect(() => {
 21 |     console.log(Boolean(meshRef.current))
 22 |   }, [])
 23 | 
 24 |   return (
 25 |     <mesh {...props} ref={meshRef}>
 26 |       <boxGeometry />
 27 |       <meshBasicMaterial />
 28 |     </mesh>
 29 |   )
 30 | }
 31 | ```
 32 | 
 33 | The exclamation mark is a non-null assertion that will let TS know that `ref.current` is defined when we access it in effects, useFrame et al. You do not need to check against null, the element is assumed to exist.
 34 | 
 35 | ## Accessing typed three-elements
 36 | 
 37 | Whenever you want to spread props or type components that rely on three elements, you can use the `ThreeElements` interface to extract the mesh, group, or any other three element, including custom elements.
 38 | 
 39 | ```tsx
 40 | import { ThreeElements } from '@react-three/fiber'
 41 | 
 42 | type FooProps = ThreeElements['mesh'] & { bar: boolean }
 43 | 
 44 | function Foo({ bar, ...props}: FooProps) {
 45 |   useEffect(() => {
 46 |     console.log(bar)
 47 |   }, [bar])
 48 |   return <mesh {...props} />
 49 | }
 50 | ```
 51 | 
 52 | ## Extend usage
 53 | 
 54 | react-three-fiber can also accept third-party elements and extend them into its internal catalogue.
 55 | 
 56 | ```tsx
 57 | import { useRef, useEffect } from 'react'
 58 | import { GridHelper } from 'three'
 59 | import { extend } from '@react-three/fiber'
 60 | 
 61 | // Create our custom element
 62 | class CustomElement extends GridHelper {}
 63 | 
 64 | // Extend so the reconciler will learn about it
 65 | extend({ CustomElement })
 66 | 
 67 | <customElement />
 68 | ```
 69 | 
 70 | The catalogue teaches the underlying reconciler how to create fibers for these elements and treat them within the scene.
 71 | 
 72 | You can then declaratively create custom elements with primitives, but TypeScript won't know about them nor their props.
 73 | 
 74 | ```html
 75 | // error: 'customElement' does not exist on type 'JSX.IntrinsicElements'
 76 | 
 77 | <customElement />
 78 | ```
 79 | 
 80 | ### Extending ThreeElements
 81 | 
 82 | To define our element in JSX, we'll use the `ThreeElement` interface to extend `ThreeElements`. This interface describes three.js classes that are available in the R3F catalog and can be used as native elements.
 83 | 
 84 | ```tsx
 85 | import { useRef, useEffect } from 'react'
 86 | import { GridHelper } from 'three'
 87 | import { extend, ThreeElement } from '@react-three/fiber'
 88 | 
 89 | // Create our custom element
 90 | class CustomElement extends GridHelper {}
 91 | 
 92 | // Extend so the reconciler will learn about it
 93 | extend({ CustomElement })
 94 | 
 95 | // Add types to ThreeElements elements so primitives pick up on it
 96 | declare module '@react-three/fiber' {
 97 |   interface ThreeElements {
 98 |     customElement: ThreeElement<typeof CustomElement>
 99 |   }
100 | }
101 | 
102 | // react-three-fiber will create your custom component and TypeScript will understand it
103 | <customComponent />
104 | ```
105 | 
106 | You can shorten element definition by using the `extend` factory signature, which will automatically extend the element locally. This will also prevent namespace bleeding.
107 | 
108 | ```tsx
109 | // Create our custom element
110 | class CustomElement extends GridHelper {}
111 | 
112 | // Extend so the reconciler will learn about it, types will be inferred
113 | const Element = extend(CustomElement)
114 | 
115 | // react-three-fiber will create your custom component and TypeScript will understand it
116 | <Element />
117 | ```
118 | 
119 | ## Extending three default elements
120 | 
121 | If you open your own root instead of using `<Canvas>`, you can extend the default elements with `extend`. But keep in mind that the `* as THREE` namespace contains classes, functions, numbers, strings. At the moment we suggest you use `any` or `@ts-ignore` unless you extract the exact classes you need (`extend({ Mesh, Group, ... })`).
122 | 
123 | ```tsx
124 | import * as THREE from 'three'
125 | import { extend, createRoot, events } from '@react-three/fiber'
126 | 
127 | // Register the THREE namespace as native JSX elements.
128 | extend(THREE as any)
129 | 
130 | // Create a react root
131 | const root = createRoot(document.querySelector('canvas'))
132 | ```
133 | 
134 | ## Exported types
135 | 
136 | react-three-fiber is extensible and exports types for its internals, such as render props, canvas props, and events:
137 | 
138 | ```tsx
139 | // Event raycaster intersection
140 | Intersection
141 | 
142 | // `useFrame` internal subscription and render callback
143 | Subscription
144 | RenderCallback
145 | 
146 | // `useThree`'s returned internal state
147 | RootState
148 | Performance
149 | Dpr
150 | Size
151 | Viewport
152 | Camera
153 | 
154 | // Canvas props
155 | CanvasProps
156 | 
157 | // Supported events
158 | Events
159 | 
160 | // Event manager signature (is completely modular)
161 | EventManager
162 | 
163 | // Wraps a platform event as it's passed through the event manager

 45 | function Portal({
 46 |   children,
 47 |   scale = [1, 1, 1],
 48 |   clearColor = 'white',
 49 |   ...props
 50 | }: {
 51 |   children: React.ReactNode
 52 |   clearColor?: string
 53 | } & ThreeElements['mesh']) {
 54 |   const ref = useRef<THREE.Mesh>(null!)
 55 |   const fbo = useFBO()
 56 |   const { events } = useThree()
 57 |   // The portal will render into this scene
 58 |   const [scene] = useState(() => new THREE.Scene())
 59 |   // We have our own camera in here, separate from the default
 60 |   const [camera] = useState(() => new THREE.PerspectiveCamera(50, 1, 0.1, 1000))
 61 | 
 62 |   useLayoutEffect(() => {
 63 |     camera.aspect = ref.current.scale.x / ref.current.scale.y
 64 |     camera.updateProjectionMatrix()
 65 |   }, [scale])
 66 | 
 67 |   useFrame((state) => {
 68 |     // Copy the default cameras whereabouts
 69 |     camera.position.copy(state.camera.position)
 70 |     camera.rotation.copy(state.camera.rotation)
 71 |     camera.scale.copy(state.camera.scale)
 72 |     // Render into a WebGLRenderTarget as a texture (the FBO above)
 73 |     state.gl.clearColor()
 74 |     state.gl.setRenderTarget(fbo)
 75 |     state.gl.render(scene, camera)
 76 |     state.gl.setRenderTarget(null)
 77 |   })
 78 | 
 79 |   // This is a custom raycast-compute function, it controls how the raycaster functions.
 80 |   const compute = useCallback<ComputeFunction>((event, state, previous) => {
 81 |     if (!previous) return false
 82 |     // First we call the previous state-onion-layers compute, this is what makes it possible to nest portals
 83 |     if (!previous.raycaster.camera) previous.events.compute!(event, previous, previous?.previousRoot?.getState())
 84 |     // We run a quick check against the textured plane itself, if it isn't hit there's no need to raycast at all
 85 |     const [intersection] = previous?.raycaster.intersectObject(ref.current)
 86 |     if (!intersection) return false
 87 |     // We take that hits uv coords, set up this layers raycaster, et voilà, we have raycasting with perspective shift
 88 |     const uv = intersection.uv!
 89 |     state.raycaster.setFromCamera(state.pointer.set(uv.x * 2 - 1, uv.y * 2 - 1), camera)
 90 |   }, [])
 91 | 
 92 |   return (
 93 |     <>
 94 |       {/* This mesh receives the render-targets texture and draws it onto a plane */}
 95 |       <mesh scale={scale} ref={ref} {...props}>
 96 |         <planeGeometry />
 97 |         <meshBasicMaterial map={fbo.texture} map-encoding={THREE.SRGBColorSpace} />
 98 |       </mesh>
 99 |       {/* A portal by default now has its own state, separate from the root state.
100 |           The third argument to createPortal allows you to override parts of it, in here for example
101 |           we place our own camera and override the events definition with a lower priority than
102 |           the previous layer, and our custom compute function. */}
103 |       {createPortal(children, scene, { camera, events: { compute, priority: events.priority - 1 } })}
104 |     </>
105 |   )
